章节列表下拉,显示所有章节的三态(有正文/仅有计划/空章),点跳章。数据来源:现有 `planningState.bookPlanning` 里的章节信息 + `chapterContext.latestChapterLabel`。P1 阶段简化:用 `planningState.currentBook` 的章节数推算列表,后续 P2 接 `/books/:id/chapter-plans` 列表接口补三态。

- [ ] **Step 1: 创建 ChapterList.jsx**

Create `frontend-react/src/components/workbench/ChapterList.jsx`:

```jsx
import { useState, useRef, useEffect } from 'react';

export default function ChapterList({ currentChapter, totalChapters, chapterNames, onSelect, onClose }) {
  const [open, setOpen] = useState(false);
  const ref = useRef(null);

  useEffect(() => {
    if (!open) return;
    function handleClickOutside(e) {
      if (ref.current && !ref.current.contains(e.target)) {
        setOpen(false);
      }
    }
    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, [open]);

  const items = Array.from({ length: totalChapters }, (_, i) => {
    const num = i + 1;
    const name = chapterNames?.[num] || '';
    // P1 简化三态:当前章之前的默认 has-content,之后默认 empty
    // P2 会接 chapter-plans 列表接口做精确三态
    const hasContent = num <= currentChapter;
    const planOnly = false;
    return { num, name, hasContent, planOnly };
  });

  return (
    <div ref={ref} style={{ position: 'relative' }}>
      <button
        type="button"
        className="workbench-global-bar-nav-btn"
        onClick={() => { setOpen((v) => !v); if (onClose) onClose(); }}
      >
        章节列表 ▾
      </button>
      {open && (
        <div className="chapter-list-dropdown">
          {items.map((item) => (
            <div
              key={item.num}
              className={`chapter-list-item${item.num === currentChapter ? ' is-current' : ''}`}
              onClick={() => { onSelect(item.num); setOpen(false); }}
            >
              <span className={`chapter-list-dot ${item.hasContent ? 'has-content' : item.planOnly ? 'plan-only' : 'empty'}`} />
              <span>第 {item.num} 章{item.name ? ` · ${item.name}` : ''}</span>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}
```

export default function CharacterDrawer({ characters, appearingRoles }) {

    <div className="drawer-character">
